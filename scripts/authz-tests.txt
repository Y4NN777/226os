# Mounted, implemented, NON-public routes mapped to an existing negative authz Go test (negative-authz gate, B1).
# Format: "METHOD /path TestName" (chi route pattern incl. the /api/v1 prefix, then the name of a Go test
# function). Comments start with '#'.
#
# Convention: TestName is TestAuthz<OperationID>... (for example TestAuthzCreateQuestionAnonymous), a
# func(*testing.T) in a test file the Go toolchain builds. scripts/check-authz-routes.sh checks that
# the test exists, does not call t.Skip (a skip inside `if testing.Short()` is allowed), contains a
# string equal to the route path (path parameters as %s or {name}) or uses the operation name, and
# mentions a denial status (401, 403 or 404) in the same declaration. It also checks that every
# mounted route is in exactly one of:
#   scripts/public-routes.txt, scripts/unimplemented-routes.txt, scripts/authz-tests.txt.
# The test must reject unauthenticated and wrong-role requests for that route (AUTHORIZATION_MATRIX).
# Slices add their lines in their own PR.
POST /api/v1/auth/sign-out TestAuthzSignOutAnonymous   # member only: anonymous callers get 401
GET /api/v1/me TestAuthzGetMeAnonymous   # member only: anonymous callers get 401
PATCH /api/v1/me/profile TestAuthzUpdateMyProfileAnonymous   # member only + CSRF: anonymous callers get 401
POST /api/v1/resources/{resourceId}/error-reports TestAuthzCreateResourceErrorReportAnonymous   # member with a verified email: anonymous 401, unverified 403
POST /api/v1/learn/articles TestAuthzCreateLearnArticleAnonymous   # member + CSRF: anonymous 401 (also for a draft), unverified member publishing 403, nothing created
POST /api/v1/projects TestAuthzCreateProjectAnonymous   # member with a verified email + CSRF: anonymous 401, unverified 403, nothing created, GitHub never called
PATCH /api/v1/projects/{projectId} TestAuthzUpdateProjectNonOwner   # owner only (C1): anonymous 401, non-owner and moderator 403 on a public project and 404 on a private one, nothing written
POST /api/v1/projects/{projectId}/contribution-needs TestAuthzCreateContributionNeedNonOwner   # owner only (C1): anonymous 401, unverified 403, non-owner 403 public / 404 private, nothing created
POST /api/v1/contribution-needs/{needId}/applications TestAuthzCreateApplicationAnonymous   # verified member, not the project owner: anonymous 401, unverified 403, someone else's private project 404, own project 409, nothing created
GET /api/v1/projects/{projectId}/applications TestAuthzListProjectApplicationsNonOwner   # owner only (C1): anonymous 401, third party and applicant 403 on a public project, 404 on a private one; no item, no count
PATCH /api/v1/applications/{applicationId} TestAuthzDecideApplicationNonParty   # owner accepts/declines, applicant withdraws: anonymous 401, third party 404 (IDOR), applicant accepting 403, owner withdrawing 403
POST /api/v1/assistant/answers TestAuthzAskAssistantAnonymous   # verified member + CSRF: anonymous 401, unverified 403, provider never called, 429 on call 11, 503 when off or down
POST /api/v1/questions TestAuthzCreateQuestionAnonymous   # verified member + CSRF: anonymous 401, unverified 403, nothing created
PATCH /api/v1/questions/{questionId} TestAuthzUpdateQuestionNonOwner   # owner only: anonymous 401, non-owner 403 on a visible question whatever the If-Match, unknown or removed 404, nothing written
POST /api/v1/questions/{questionId}/answers TestAuthzCreateAnswerAnonymous   # verified member + CSRF: anonymous 401, unverified 403, unknown or removed question 404, nothing created
POST /api/v1/votes TestAuthzCastVoteAnonymous   # verified member + CSRF, not own content: anonymous 401, unverified 403, unknown or removed target 404, own content 403 SELF_VOTE_FORBIDDEN, no vote written
DELETE /api/v1/votes TestAuthzRetractVoteAnonymous   # verified member + CSRF: anonymous 401, unverified 403, unknown or removed target 404, nothing written
POST /api/v1/questions/{questionId}/accept TestAuthzAcceptAnswerNonOwner   # question author only: anonymous 401, unverified 403, non-author 403 and nothing changes, unknown or removed question 404, answer of another question 422
POST /api/v1/questions/{questionId}/unaccept TestAuthzUnacceptAnswerNonOwner   # question author only: anonymous 401, unverified 403, non-author 403 and nothing changes, unknown or removed question 404
