// Route-table gate (G4, decision B1). NOT part of the module build: it is compiled into the httpx
// test binary through `go test -overlay` by scripts/check-authz-routes.sh, so it can reach the
// internal router without touching application source. Kept as .txt so that gofmt, go vet and
// golangci-lint over ./... never see it as a stray package.
//
// Rules, all fail-closed:
//  1. every route mounted by httpx.NewRouter is in EXACTLY ONE of public-routes.txt,
//     unimplemented-routes.txt, authz-tests.txt;
//  2. every test named in authz-tests.txt exists (the candidate names come from `go test -list`),
//     is named TestAuthz<OperationID>... for a contract operation (TestAuthz... for any other route),
//     and does not call t.Skip, t.SkipNow or t.Skipf directly;
//  3. public and authz entries must be mounted routes; unimplemented entries must be operations of
//     the contract (apigen route table) so a typo cannot hide a route;
//  4. no router, mux or route registration exists outside httpx.NewRouter (go/ast scan), and the
//     NotFound and MethodNotAllowed handlers inside it only call writeProblem (they cannot serve a path);
//  5. the middleware chain installed by httpx.NewRouter equals router-middleware.txt: a middleware can
//     answer a path that has no route (invisible to chi.Walk), so each one is a security review item.
//
// Residual, by design: what a mapped test actually asserts, and a skip hidden behind a helper, are
// review matters. The scan is syntactic.
package httpx_test

const (
	filePublic        = "public-routes.txt"
	fileUnimplemented = "unimplemented-routes.txt"
	fileAuthz         = "authz-tests.txt"
	fileMiddleware    = "router-middleware.txt"
	apiPrefix         = "/api/v1"
	authzTestPrefix   = "TestAuthz"
)

type entry struct{ route, test string }

func TestAuthzRouteTable(t *testing.T) {
	dir := mustEnv(t, "AUTHZ_LISTS_DIR")
	public := readList(t, filepath.Join(dir, filePublic), 2)
	unimplemented := readList(t, filepath.Join(dir, fileUnimplemented), 2)
	authz := readList(t, filepath.Join(dir, fileAuthz), 3)
	testNames := readNames(t, mustEnv(t, "AUTHZ_TEST_NAMES"))

	handler := httpx.NewRouter(slog.New(slog.NewTextHandler(io.Discard, nil)))
	mounted := walk(t, handler)
	if len(mounted) == 0 {
		t.Fatal("authz-gate: the router mounts no route at all; the check cannot prove anything")
	}
	contract := map[string]bool{}
	for _, r := range walk(t, apigen.Handler(apigen.Unimplemented{})) {
		method, path, _ := strings.Cut(r, " ")
		contract[method+" "+apiPrefix+path] = true
	}
	if len(contract) == 0 {
		t.Fatal("authz-gate: the generated server mounts no operation; the contract set is empty")
	}

	where := map[string][]string{}
	for file, list := range map[string][]entry{filePublic: public, fileUnimplemented: unimplemented, fileAuthz: authz} {
		for _, e := range list {
			where[e.route] = append(where[e.route], file)
		}
	}
	isMounted := map[string]bool{}
	for _, r := range mounted {
		isMounted[r] = true
	}
	t.Logf("authz-gate: mounted routes: %d, public: %d, unimplemented: %d, authz-tested: %d",
		len(mounted), len(public), len(unimplemented), len(authz))
	for _, r := range mounted {
		t.Logf("route %s lists=%v", r, where[r])
	}

	for _, r := range mounted {
		switch n := len(where[r]); {
		case n == 0:
			t.Errorf("authz-gate: route %q is mounted but is in none of scripts/%s, scripts/%s, scripts/%s: "+
				"add it to the public list with a reason, or write its negative authz test and map it in %s",
				r, filePublic, fileUnimplemented, fileAuthz, fileAuthz)
		case n > 1:
			t.Errorf("authz-gate: route %q is listed in more than one of the three lists (%v): it must be in exactly one", r, where[r])
		}
	}
	for _, e := range public {
		if !isMounted[e.route] {
			t.Errorf("authz-gate: %s lists %q but that route is not mounted", filePublic, e.route)
		}
	}
	roots := map[string]bool{mustEnv(t, "AUTHZ_SCAN_ROOT"): true}
	if extra := os.Getenv("AUTHZ_SCAN_EXTRA"); extra != "" {
		roots[extra] = false
	}
	skipping := skippingTests(t, roots)
	opIDs := operationIDs(t, mustEnv(t, "AUTHZ_OPENAPI"))
	for _, e := range authz {
		want := authzTestPrefix
		if id, ok := opIDs[e.route]; ok {
			want += strings.ToUpper(id[:1]) + id[1:]
		}
		if !strings.HasPrefix(e.test, want) {
			t.Errorf("authz-gate: test %q for route %q must start with %s (a negative authz test named after the operation)", e.test, e.route, want)
		}
		if pos, ok := skipping[e.test]; ok {
			t.Errorf("authz-gate: authz test %q for route %q skips itself (%s): a skipped test proves nothing", e.test, e.route, pos)
		}
		if !isMounted[e.route] {
			t.Errorf("authz-gate: %s lists %q but that route is not mounted", fileAuthz, e.route)
		}
		if !testNames[e.test] {
			t.Errorf("authz-gate: authz test %q for route %q does not exist (go test -list)", e.test, e.route)
		}
	}
	// Unimplemented entries need not be mounted yet (the mount lands in api-mount); they must be real
	// operations, and a mounted one must really answer 501: a line left here after a slice implemented
	// the route would otherwise exempt it from the authz-test requirement.
	for _, e := range unimplemented {
		if !contract[e.route] {
			t.Errorf("authz-gate: %q in %s is not an operation of the contract (api/openapi.yaml)", e.route, fileUnimplemented)
		}
		if isMounted[e.route] {
			if code := probe(handler, e.route); code != http.StatusNotImplemented {
				t.Errorf("authz-gate: route %q is listed in %s but answers %d, want 501: remove it there and add it to %s or %s",
					e.route, fileUnimplemented, code, filePublic, fileAuthz)
			}
		}
	}

	// The project tree gets the two exact exemptions (generated packages, the httpx package); an extra
	// scan root (test hook) gets none.
	for root, project := range roots {
		findings, middleware := scanOutsideNewRouter(t, root, project)
		for _, f := range findings {
			t.Errorf("authz-gate: %s: %s outside httpx.NewRouter: every router, mux and route registration must go through httpx.NewRouter", f.pos, f.what)
		}
		if project {
			checkMiddleware(t, filepath.Join(dir, fileMiddleware), middleware)
		}
	}
}

var pathParam = regexp.MustCompile(`\{[^}]+\}`)

// probe serves one request for route (path parameters filled with a UUID, no body, no session) and
// returns the status code.
func probe(h http.Handler, route string) int {
	method, path, _ := strings.Cut(route, " ")
	path = pathParam.ReplaceAllString(path, "0190c3f2-7a3b-7c1e-8a5d-3b1f2c4d5e6f")
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest(method, path, nil))
	return rec.Code
}

// skippingTests returns the test functions under the roots that call t.Skip, t.SkipNow or t.Skipf.
func skippingTests(t *testing.T, roots map[string]bool) map[string]string {
	t.Helper()
	out := map[string]string{}
	fset := token.NewFileSet()
	for root := range roots {
		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
			if err != nil {
				return err
			}
			if d.IsDir() {
				if d.Name() == "testdata" || d.Name() == "node_modules" {
					return filepath.SkipDir
				}
				return nil
			}
			if !strings.HasSuffix(path, "_test.go") {
				return nil
			}
			file, perr := parser.ParseFile(fset, path, nil, 0)
			if perr != nil {
				return fmt.Errorf("parse %s: %w", path, perr)
			}
			for _, decl := range file.Decls {
				fn, ok := decl.(*ast.FuncDecl)
				if !ok || fn.Recv != nil || fn.Body == nil {
					continue
				}
				ast.Inspect(fn.Body, func(n ast.Node) bool {
					if call, ok := n.(*ast.CallExpr); ok {
						if sel, ok := call.Fun.(*ast.SelectorExpr); ok && strings.HasPrefix(sel.Sel.Name, "Skip") {
							out[fn.Name.Name] = fset.Position(call.Pos()).String()
						}
					}
					return true
				})
			}
			return nil
		})
		if err != nil {
			t.Fatalf("authz-gate: scan tests under %s: %v", root, err)
		}
	}
	return out
}

// operationIDs maps "METHOD /api/v1/path" to the operationId of api/openapi.yaml.
func operationIDs(t *testing.T, path string) map[string]string {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil {
		t.Fatalf("authz-gate: read contract: %v", err)
	}
	var doc struct {
		Paths map[string]map[string]any `yaml:"paths"`
	}
	if err := yaml.Unmarshal(b, &doc); err != nil {
		t.Fatalf("authz-gate: parse contract: %v", err)
	}
	out := map[string]string{}
	for p, item := range doc.Paths {
		for method, op := range item {
			if m, ok := op.(map[string]any); ok {
				if id, ok := m["operationId"].(string); ok && id != "" {
					out[strings.ToUpper(method)+" "+apiPrefix+p] = id
				}
			}
		}
	}
	if len(out) == 0 {
		t.Fatal("authz-gate: the contract has no operationId")
	}
	return out
}

func mustEnv(t *testing.T, key string) string {
	t.Helper()
	v := os.Getenv(key)
	if v == "" {
		t.Fatalf("authz-gate: %s is not set; refusing to run without it", key)
	}
	return v
}

func walk(t *testing.T, h http.Handler) []string {
	t.Helper()
	router, ok := h.(chi.Routes)
	if !ok {
		t.Fatal("authz-gate: the handler is not a chi router; the route table cannot be listed")
	}
	var routes []string
	if err := chi.Walk(router, func(method, route string, _ http.Handler, _ ...func(http.Handler) http.Handler) error {
		routes = append(routes, method+" "+route)
		return nil
	}); err != nil {
		t.Fatalf("authz-gate: walk route table: %v", err)
	}
	sort.Strings(routes)
	return routes
}

func readNames(t *testing.T, path string) map[string]bool {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil {
		t.Fatalf("authz-gate: read test names: %v", err)
	}
	names := map[string]bool{}
	for _, n := range strings.Fields(string(b)) {
		names[n] = true
	}
	return names
}

// readList parses a route list: "METHOD /path" (fields == 2) or "METHOD /path TestName" (fields == 3).
// A malformed line is an error, never skipped. A route repeated in one file yields two entries.
func readList(t *testing.T, path string, fields int) []entry {
	t.Helper()
	f, err := os.Open(path)
	if err != nil {
		t.Fatalf("authz-gate: BLOCKED: read %s: %v", path, err)
	}
	defer f.Close()
	var out []entry
	sc := bufio.NewScanner(f)
	for n := 1; sc.Scan(); n++ {
		line := sc.Text()
		if i := strings.Index(line, "#"); i >= 0 {
			line = line[:i]
		}
		parts := strings.Fields(line)
		if len(parts) == 0 {
			continue
		}
		if len(parts) != fields || parts[0] != strings.ToUpper(parts[0]) || !strings.HasPrefix(parts[1], "/") {
			want := `"METHOD /path"`
			if fields == 3 {
				want = `"METHOD /path TestName"`
			}
			t.Errorf("authz-gate: %s:%d: malformed line %q (want %s)", filepath.Base(path), n, strings.TrimSpace(line), want)
			continue
		}
		e := entry{route: parts[0] + " " + parts[1]}
		if fields == 3 {
			e.test = parts[2]
		}
		out = append(out, e)
	}
	if err := sc.Err(); err != nil {
		t.Fatalf("authz-gate: read %s: %v", path, err)
	}
	return out
}

type finding struct{ pos, what string }

var (
	routeMethods = map[string]bool{
		"Get": true, "Post": true, "Put": true, "Patch": true, "Delete": true, "Head": true, "Options": true,
		"Connect": true, "Trace": true, "Handle": true, "HandleFunc": true, "Mount": true, "Route": true,
		"Method": true, "MethodFunc": true, "NotFound": true, "MethodNotAllowed": true, "Use": true, "With": true,
	}
	patternArg = regexp.MustCompile(`^(/|(GET|POST|PUT|PATCH|DELETE|HEAD|OPTIONS|CONNECT|TRACE)( |$))`)
	// builders[importPath] lists the functions that build a router, mux or server entry point.
	builders = map[string]map[string]bool{
		"github.com/go-chi/chi/v5": {"NewRouter": true, "NewMux": true},
		"net/http": {
			"NewServeMux": true, "Handle": true, "HandleFunc": true, "ListenAndServe": true,
			"ListenAndServeTLS": true, "Serve": true, "ServeTLS": true, "FileServer": true, "FileServerFS": true,
		},
		"net/http/fcgi": {"Serve": true},
		"github.com/Y4NN777/226os/src/api/internal/platform/apigen": {
			"Handler": true, "HandlerFromMux": true, "HandlerFromMuxWithBaseURL": true, "HandlerWithOptions": true,
		},
	}
	// routerTypes[importPath] lists the types through which routes can be registered.
	routerTypes = map[string]map[string]bool{
		"github.com/go-chi/chi/v5": {"Router": true, "Mux": true, "Routes": true},
		"net/http":                 {"ServeMux": true},
	}

// scanOutsideNewRouter parses the non-test, non-generated Go files under root and reports every
// router builder, router-typed reference and route registration outside the NewRouter function of
// the httpx package. The heuristics are syntactic (patterns given as string literals, router types
// named through their package); anything they cannot see is still caught by chi.Walk on the mounted table.
func scanOutsideNewRouter(t *testing.T, root string, project bool) ([]finding, []string) {
	t.Helper()
	var out []finding
	var middleware []string
	fset := token.NewFileSet()
	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		rel, _ := filepath.Rel(root, path)
		rel = filepath.ToSlash(rel)
		if d.IsDir() {
			// Exact generated packages only (not any directory that happens to carry the name), plus
			// directories the Go toolchain itself ignores.
			if n := d.Name(); n == "node_modules" || n == "testdata" {
				return filepath.SkipDir
			}
			return nil
		}
		if !strings.HasSuffix(path, ".go") || strings.HasSuffix(path, "_test.go") {
			return nil
		}
		var src any // nil: read the file. Test hook: the override replaces router.go for the scan too.
		if ov := os.Getenv("AUTHZ_ROUTER_OVERRIDE"); ov != "" && project && rel == "internal/platform/httpx/router.go" {
			b, rerr := os.ReadFile(ov)
			if rerr != nil {
				return rerr
			}
			src = b
		}
		file, perr := parser.ParseFile(fset, path, src, parser.ParseComments)
		if perr != nil {
			return fmt.Errorf("parse %s: %w", path, perr)
		}
		if project && isGeneratedDir(rel) && isGenerated(file) {
			return nil // sqlc and oapi-codegen output; hand edits there fail `make generate` drift (G1)
		}
		inHTTPX := project && filepath.ToSlash(filepath.Dir(rel)) == "internal/platform/httpx"
		found, mw := scanFile(fset, file, inHTTPX)
		out = append(out, found...)
		middleware = append(middleware, mw...)
		return nil
	})
	if err != nil {
		t.Fatalf("authz-gate: scan %s: %v", root, err)
	}
	return out, middleware
}

// checkMiddleware compares the middleware chain installed by NewRouter (source text of every Use and
// With argument, in source order) with the reviewed list.
func checkMiddleware(t *testing.T, path string, installed []string) {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil {
		t.Fatalf("authz-gate: BLOCKED: read %s: %v", path, err)
	}
	var listed []string
	for _, line := range strings.Split(string(b), "\n") {
		if i := strings.Index(line, "#"); i >= 0 {
			line = line[:i]
		}
		if line = strings.TrimSpace(line); line != "" {
			listed = append(listed, line)
		}
	}
	inList := map[string]bool{}
	for _, m := range listed {
		inList[m] = true
	}
	inRouter := map[string]bool{}
	for _, m := range installed {
		inRouter[m] = true
		if !inList[m] {
			t.Errorf("authz-gate: middleware %q is installed by httpx.NewRouter but is not in %s: a middleware can answer a path that has no route, so each one is a security review item", m, fileMiddleware)
		}
	}
	for _, m := range listed {
		if !inRouter[m] {
			t.Errorf("authz-gate: %s lists %q but httpx.NewRouter does not install it", fileMiddleware, m)
		}
	}
}

// isGenerated honours the standard marker only above the package clause, as the Go tools do.
func isGeneratedDir(rel string) bool {
	dir := filepath.ToSlash(filepath.Dir(rel))
	return dir == "internal/platform/apigen" || dir == "internal/platform/dbgen"
}

func isGenerated(f *ast.File) bool {
	for _, cg := range f.Comments {
		if cg.Pos() > f.Package {
			break
		}
		for _, c := range cg.List {
			if strings.HasPrefix(c.Text, "// Code generated") && strings.Contains(c.Text, "DO NOT EDIT") {
				return true
			}
		}
	}
	return false
}

func scanFile(fset *token.FileSet, file *ast.File, inHTTPX bool) ([]finding, []string) {
	imports := map[string]string{} // local name -> import path
	for _, imp := range file.Imports {
		p, _ := strconv.Unquote(imp.Path.Value)
		name := filepath.Base(p)
		if p == "github.com/go-chi/chi/v5" {
			name = "chi"
		}
		if imp.Name != nil {
			name = imp.Name.Name
		}
		imports[name] = p
	}
	var out []finding
	var middleware []string
	report := func(n ast.Node, what string) {
		out = append(out, finding{pos: fset.Position(n.Pos()).String(), what: what})
	}
	for _, imp := range file.Imports {
		if p, _ := strconv.Unquote(imp.Path.Value); imp.Name != nil && imp.Name.Name == "." && (builders[p] != nil || routerTypes[p] != nil) {
			report(imp, "dot import of "+p+" hides router builders")
		}
	}
	for _, decl := range file.Decls {
		if fn, ok := decl.(*ast.FuncDecl); ok {
			if inHTTPX && fn.Recv == nil && fn.Name.Name == "NewRouter" {
				// The one function allowed to build and register. Its NotFound and MethodNotAllowed
				// handlers still must not answer a path themselves: chi.Walk cannot see that.
				ast.Inspect(fn, func(n ast.Node) bool {
					call, ok := n.(*ast.CallExpr)
					if !ok {
						return true
					}
					sel, ok := call.Fun.(*ast.SelectorExpr)
					if !ok {
						return true
					}
					switch sel.Sel.Name {
					case "NotFound", "MethodNotAllowed":
						if !onlyWriteProblem(call.Args) {
							report(call, sel.Sel.Name+" handler must contain only writeProblem")
						}
					case "Use", "With":
						for _, a := range call.Args {
							middleware = append(middleware, types.ExprString(a))
						}
					}
					return true
				})
				continue
			}
			if fn.Recv != nil && fn.Name.Name == "ServeHTTP" {
				report(fn, "a type implements http.Handler (ServeHTTP)")
			}
		}
		ast.Inspect(decl, func(n ast.Node) bool {
			switch x := n.(type) {
			case *ast.SelectorExpr:
				id, ok := x.X.(*ast.Ident)
				if !ok {
					return true
				}
				path := imports[id.Name]
				if builders[path][x.Sel.Name] {
					report(x, id.Name+"."+x.Sel.Name+" builds a router, mux or handler")
				}
				if routerTypes[path][x.Sel.Name] {
					report(x, id.Name+"."+x.Sel.Name+" (router type) is used")
				}
				_ = path
			case *ast.CompositeLit:
				if sel, ok := x.Type.(*ast.SelectorExpr); ok && !inHTTPX {
					if id, ok := sel.X.(*ast.Ident); ok && imports[id.Name] == "net/http" && sel.Sel.Name == "Server" {
						report(x, id.Name+".Server (a server wrapping a handler) is built")
					}
				}
			case *ast.InterfaceType:
				// Only the registration SHAPE: a route-registration method name whose signature takes an
				// http.Handler or http.HandlerFunc (so repository interfaces with Get/Delete are fine).
				for _, m := range x.Methods.List {
					ft, ok := m.Type.(*ast.FuncType)
					if !ok {
						continue
					}
					for _, name := range m.Names {
						if routeMethods[name.Name] && mentionsHandler(ft, imports) {
							report(m, "an interface declares the route registration method "+name.Name)
						}
					}
				}
			case *ast.CallExpr:
				if sel, ok := x.Fun.(*ast.SelectorExpr); ok && routeMethods[sel.Sel.Name] && isRegistration(sel.Sel.Name, x.Args) {
					report(x, "route registration ."+sel.Sel.Name+"(...)")
				}
			}
			return true
		})
	}
	return out, middleware
}

// mentionsHandler reports whether a method signature names net/http's Handler, HandlerFunc or a
// function over them (the middleware shape).
func mentionsHandler(ft *ast.FuncType, imports map[string]string) bool {
	found := false
	ast.Inspect(ft, func(n ast.Node) bool {
		if sel, ok := n.(*ast.SelectorExpr); ok {
			if id, ok := sel.X.(*ast.Ident); ok && imports[id.Name] == "net/http" && (sel.Sel.Name == "Handler" || sel.Sel.Name == "HandlerFunc") {
				found = true
			}
		}
		return !found
	})
	return found
}

// onlyWriteProblem reports whether args is one function literal whose body is a single call to writeProblem.
func onlyWriteProblem(args []ast.Expr) bool {
	if len(args) != 1 {
		return false
	}
	lit, ok := args[0].(*ast.FuncLit)
	if !ok || len(lit.Body.List) != 1 {
		return false
	}
	stmt, ok := lit.Body.List[0].(*ast.ExprStmt)
	if !ok {
		return false
	}
	call, ok := stmt.X.(*ast.CallExpr)
	if !ok {
		return false
	}
	id, ok := call.Fun.(*ast.Ident)
	return ok && id.Name == "writeProblem"
}

// isRegistration recognises a registration call by its arguments: a literal pattern first and a
// handler after it (chi: Get("/x", h), Handle("GET /x", h), Mount("/x", h); Method("GET", "/x", h)).
// Client calls such as c.Get(ctx, "/path") do not match.
func isRegistration(name string, args []ast.Expr) bool {
	if len(args) < 2 {
		return false
	}
	lit := func(e ast.Expr) (string, bool) {
		b, ok := e.(*ast.BasicLit)
		if !ok || b.Kind != token.STRING {
			return "", false
		}
		v, err := strconv.Unquote(b.Value)
		return v, err == nil
	}
	if name == "Method" || name == "MethodFunc" {
		if m, ok := lit(args[0]); ok && patternArg.MatchString(m) {
			return true
		}
	}
	v, ok := lit(args[0])
	return ok && patternArg.MatchString(v)
}
